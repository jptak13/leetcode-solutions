// Title: Kth Missing Positive Number
            // Difficulty: Unknown
            // Language: Python3
            // Link: https://leetcode.com/problems/kth-missing-positive-number/

class Solution:
    def findKthPositive(self, arr: list[int], k: int) -> int:
        last = i = curr = 0
        while i < len(arr):
            curr = arr[i] - last - 1
            if curr >= k:
                return k + last
            k -= curr
            last = arr[i]
            i += 1
        return last + k
