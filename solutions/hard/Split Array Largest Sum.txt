// Title: Split Array Largest Sum
            // Difficulty: Hard
            // Language: Python3
            // Link: https://leetcode.com/problems/split-array-largest-sum/

class Solution:
    def splitArray(self, nums: list[int], k: int) -> int:
        l = max(nums)
        r = sum(nums)
        while l < r:
            mid = (l + r) // 2
            i = 0
            total = 0
            while i < len(nums):
                curr = 0
                while i < len(nums) and curr + nums[i] <= mid:
                    curr += nums[i]
                    i += 1
                total += 1
            if total > k:
                l = mid + 1
            else:
                r = mid
        return r
