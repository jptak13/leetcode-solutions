// Title: Pascal's Triangle
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/pascal's-triangle/

class Solution:
    def generate(self, numRows: int) -> list[list[int]]:
        nums = [[1], [1, 1]]
        if numRows == 1:
            return [[1]]
        elif numRows == 2:
        for i in range(2, numRows):
            nums.append([1])
            for j in range(i - 1):
            return nums
                nums[i].append(nums[i - 1][j] + nums[i - 1][j + 1])
            nums[i].append(1)
        return nums
