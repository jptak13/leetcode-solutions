// Title: Climbing Stairs
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/climbing-stairs/

class Solution:
    def climbStairs(self, n: int) -> int:
        x = 0
        y = 1
        for i in range(n):
            tmp = x
            x = y
            y += tmp
        return y
