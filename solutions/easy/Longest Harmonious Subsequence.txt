// Title: Longest Harmonious Subsequence
            // Difficulty: Easy
            // Language: Python3
            // Link: https://leetcode.com/problems/longest-harmonious-subsequence/

class Solution:
    def findLHS(self, nums: list[int]) -> int:
        x = {}
        out = 0
        for num in nums:
            if num not in x:
                x[num] = 1
            else:
                x[num] += 1
        for key in x:
            if key + 1 in x:
               out = max(out, x[key] + x[key + 1]) 
        return out
