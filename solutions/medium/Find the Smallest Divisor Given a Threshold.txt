// Title: Find the Smallest Divisor Given a Threshold
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/find-the-smallest-divisor-given-a-threshold/

class Solution:
    def smallestDivisor(self, nums: list[int], threshold: int) -> int:
        l = 1
        r = max(nums)
        while l < r:
            mid = (l + r) // 2
            curr = 0
            for num in nums:
                curr += (num + mid - 1) // mid
            if curr <= threshold:
                r = mid
            else:
                l = mid + 1
        return l
