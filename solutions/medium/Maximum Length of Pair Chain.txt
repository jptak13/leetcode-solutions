// Title: Maximum Length of Pair Chain
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/maximum-length-of-pair-chain/

class Solution:
    def findLongestChain(self, pairs: list[list[int]]) -> int:
        curr = most = 0
        pairs.sort(key=lambda x: x[1])
        last = float('-inf')
        for start, end in pairs:
            if start > last:
                last = end
                curr += 1
                most = max(most, curr)
        return most
