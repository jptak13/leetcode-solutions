// Title: Jump Game
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/jump-game/

class Solution:
    def canJump(self, nums: list[int]) -> bool:
        target = len(nums) - 1
        for i in range(len(nums) - 1, -1, -1):
            if i + nums[i] >= target:
                target = i
        if len(nums) == 1:
            return True
        return target == 0
