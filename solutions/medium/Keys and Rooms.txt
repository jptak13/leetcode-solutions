// Title: Keys and Rooms
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/keys-and-rooms/

class Solution:
    def canVisitAllRooms(self, rooms: list[list[int]]) -> bool:
        start = 0
        seen = {0: 1}
        queue = [0]
        visited = 1
        while start < len(queue):
            end = len(queue)
            while start < end:
                for i in range(len(rooms[queue[start]])):
                    if rooms[queue[start]][i] not in seen:
                        seen[rooms[queue[start]][i]] = 1
                        queue.append(rooms[queue[start]][i])
                        visited += 1
                start += 1
            print(start)
            print(queue)
        print(seen)
        print(visited)
