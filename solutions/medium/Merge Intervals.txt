// Title: Merge Intervals
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/merge-intervals/

class Solution:
    def merge(self, intervals: list[list[int]]) -> list[list[int]]:
        intervals = sorted(intervals)
        out = []
        i = j = 0
        most = float('-inf')
        print(intervals)
        while i < len(intervals):
            while j < len(intervals) and (intervals[i][1] >= intervals[j][0] or most >= intervals[j][0]):
                most = max(most, intervals[j][1])
                j += 1
            out.append([intervals[i][0], most])
            most = float('-inf')
            if i != j:
                i = j
            else:
                i += 1
        return out
