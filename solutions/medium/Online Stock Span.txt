            // Difficulty: Medium

class StockSpanner:

    def __init__(self):
        self.stack = []

    def next(self, price: int) -> int:
        if self.stack and self.stack[-1][0] <= price:
            toReturn = 1
            while self.stack and self.stack[-1][0] <= price:
                toReturn += self.stack.pop()[1]
            self.stack.append([price, toReturn])
        else:
            toReturn = 1
            self.stack.append([price, 1])
        return toReturn

# obj = StockSpanner()
